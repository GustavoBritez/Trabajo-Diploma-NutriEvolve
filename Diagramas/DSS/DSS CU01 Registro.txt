@startuml Secuencia_CU01_Multicapa
skinparam style strictuml
skinparam sequenceMessageAlign center
skinparam shadowing false
skinparam roundcorner 6
autonumber "<b>[00]</b>"

actor "Nutricionista" as Nutricionista

box "Capa GUI" #EBF5FB
    participant "GUI\n(FrmRegistrarTurno_DNI101)" as UI
end box

box "Capa BLL (Lógica de Negocio)" #E8F8F5
    participant "AgendaMedicaBLL_DNI101" as AgendaBLL
    participant "PacienteBLL_DNI101" as PacienteBLL
    participant "CUN02: Registrar Paciente" as CUN02
    participant "TurnoBLL_DNI101" as TurnoBLL
    participant "BitacoraBLL" as BitacoraBLL
    participant "DigitoVerificadorBLL" as DVBLL
end box

box "Capa BE & Patrón State" #FEF9E7
    participant "TurnoBE_DNI101" as TurnoBE
    participant "TurnoSolicitadoState_DNI101" as State
end box

box "Capa DAL (Table Data Gateway)" #FDF2E9
    participant "DAL" as DAL
end box

title Diagrama de Secuencia de Realización - CUN01: Registrar Turno (Multicapa BLL/DAL/BE)

== 1. Consulta de Bloques Horarios Disponibles ==

Nutricionista -> UI: btn_Registrar_Turno()
activate UI
UI -> UI: ConsultarDisponibilidad()
UI -> AgendaBLL: ListarBloquesDisponibles(fecha, dniNutri)
activate AgendaBLL
AgendaBLL -> DAL: ListarBloquesDisponibles(fecha, dniNutri)
activate DAL
DAL --> AgendaBLL
deactivate DAL

AgendaBLL -> AgendaBLL: MapearBloque(DataRow)\n[new BloqueHorarioBE_DNI101]
AgendaBLL -> DAL: ObtenerHorariosOcupadosPorProfesional(dniNutri, fecha)
activate DAL
DAL --> AgendaBLL
deactivate DAL

AgendaBLL --> UI
deactivate AgendaBLL
UI --> Nutricionista
deactivate UI

== 2. Identificación del Paciente (txtDniNiño_Leave) y Punto de Extensión CUN-02 ==

Nutricionista -> UI: txtDniNiño_Leave(dniNiño)
activate UI
UI -> PacienteBLL: ObtenerOAsegurarPaciente(dniNiño, callback)
activate PacienteBLL
PacienteBLL -> DAL: ObtenerPacientePorDNI(dniNiño)
activate DAL
DAL --> PacienteBLL
deactivate DAL

alt Paciente No Registrado [Punto de Extensión CUN-02 disparado por BLL]
    PacienteBLL -> CUN02: RegistrarPaciente(dniNiño)
    activate CUN02
    Nutricionista -> CUN02: IngresarDatosPaciente(nombre, apellido, dni, tel, email, obraSocial)
    CUN02 -> DAL: Guardar(nombre, apellido, dni, tel, email, obraSocial)
    activate DAL
    DAL --> CUN02
    deactivate DAL
    CUN02 --> PacienteBLL
    deactivate CUN02
else Paciente Registrado
    PacienteBLL -> PacienteBLL: MapearPaciente(dt.Rows[0])\n[new PacienteBE_DNI101]
end

PacienteBLL --> UI
deactivate PacienteBLL
UI --> Nutricionista
deactivate UI

== 3. Confirmación, Validación y Registro de Turno (Patrón State + SqlTransaction) ==

Nutricionista -> UI: btnRegistrarTurno_Click()
activate UI
UI -> UI: ValidarCamposObligatorios()
UI -> TurnoBLL: RegistrarTurno(dniNiño, fecha, horario, motivo, idBloque, dniNutri)
activate TurnoBLL

TurnoBLL -> PacienteBLL: ObtenerPacientePorDNI(dniNiño)
activate PacienteBLL
PacienteBLL --> TurnoBLL
deactivate PacienteBLL

TurnoBLL -> DAL: ExisteTurnoParaProfesional(dniNutri, fecha, hora)
activate DAL
DAL --> TurnoBLL
deactivate DAL

TurnoBLL -> TurnoBE **: new TurnoBE_DNI101(dniNiño, fecha, horario, motivo)
TurnoBLL -> TurnoBE: GenerarCodigoUnico()
TurnoBLL -> State **: new TurnoSolicitadoState_DNI101()
TurnoBLL -> TurnoBE: ConfigurarEstado(state)
TurnoBLL -> TurnoBLL: CalcularDV(cadenaDV)

TurnoBLL -> DAL: RegistrarTurnoConBloque(codigoTurno, fecha, hora, motivo, estado, idPaciente, dniNutri, idBloque, dv)\n[SqlTransaction atómica en SQL Server]
activate DAL
DAL --> TurnoBLL
deactivate DAL

TurnoBLL -> BitacoraBLL: RegistrarBitacora(1, Turno registrado..., dniNutri, TurneroNutricional)
activate BitacoraBLL
BitacoraBLL -> DAL: GuardarBitacora(bitacora)
activate DAL
DAL --> BitacoraBLL
deactivate DAL
BitacoraBLL --> TurnoBLL
deactivate BitacoraBLL

TurnoBLL -> DVBLL: RecalcularYPersistir()
activate DVBLL
DVBLL -> DAL: PersistirIntegridadCompleta(resumen, filasDV)\n[Batch Transaction SQL]
activate DAL
DAL --> DVBLL
deactivate DAL
DVBLL --> TurnoBLL
deactivate DVBLL

TurnoBLL --> UI
deactivate TurnoBLL

UI --> Nutricionista
deactivate UI

@enduml
